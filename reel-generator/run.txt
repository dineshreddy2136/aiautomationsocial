# =====================================================
# OPTIONS REFERENCE
# =====================================================
# --crop-style OPTIONS:
#   blur         = Fit video in center with blurred background (default)
#   crop-center  = Center crop to 9:16 (cuts off sides)
#   crop-top     = Focus on top of video (cuts off bottom)
#   crop-bottom  = Focus on bottom of video (cuts off top)
#
# --auto-reframe   = Basic face detection tracking
# --smart-reframe  = LLM-enhanced intelligent reframing (YOLO + Gemini Vision)
# --trim-bottom 50 = Crop 50 pixels from bottom of source (for watermarks)
# --landscape      = Keep landscape format (no vertical conversion)
