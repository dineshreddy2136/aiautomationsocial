# REEL GENERATOR - QUICK REFERENCE

# ACTIVATE VENV (run this first)
source venv/bin/activate

# FULL PIPELINE (Download → Transcript → AI Plan → Cut Videos)
python main.py --url "https://youtu.be/ygKDapeYOSc?si=trGirNJiin4ewbQw" --generate-videos

# PLAN ONLY (No video cutting)
python main.py --url "https://youtu.be/VIDEO_ID"

# CUT FROM EXISTING PLAN
python cut_reels.py --video videos/video.mp4 --plan output/FOLDER/reel_plans.json

# WITH SMART REFRAME (Recommended - uses YOLO for intelligent cropping)
python main.py --url "https://youtu.be/VIDEO_ID" --generate-videos --smart-reframe
python cut_reels.py --video videos/video.mp4 --plan output/FOLDER/reel_plans.json --smart-reframe

# OPTIONS
# --crop-style blur|crop-center|crop-top|crop-bottom
# --auto-reframe    Basic face tracking
# --smart-reframe   YOLO-based intelligent reframing (recommended)
# --trim-bottom 50  Remove watermarks (crop N pixels from bottom)
# --landscape       Keep 16:9 format instead of vertical
